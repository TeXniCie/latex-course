\begin{frame}[fragile]{Oefening}
    Maak een commando zodat \mintinline{tex}{\gem{x}{w}} het volgende weergeeft:
    \begin{align*}
        \operatorname{Gem}^y{(x)}
    \end{align*}
    Let op, $\operatorname{Gem}$ is \emph{rechtgedrukt}!
    \pause 
    \begin{tcolorbox}[colback=white, colframe=green!70!black,title={Antwoord}]
        \mintinline{tex}{\newcommand{\gem}[1]{\operatorname{Gem}^{#2}(#1)}}
    \end{tcolorbox}
\end{frame}
