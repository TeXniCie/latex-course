\begin{frame}[fragile]{Oefening}
    Maak een commando zodat \mintinline{tex}{\t(A)} het volgende 
    weergeeft\footnote{$\operatorname{Tr}$ is een notatie voor het \emph{spoor} van een matrix}:
    \begin{align*}
        \operatorname{Tr}(A)
    \end{align*}
    Let op, \mintinline{tex}{\t} is standaard al gedefinieerd! Bovendien is Tr \emph{rechtgedrukt}!
    \pause 
    \begin{tcolorbox}[colback=white, colframe=green!70!black,title={Antwoord}]
        \mintinline{tex}{\renewcommand{\t}{\operatorname{Tr}}}

        \t(A)
    \end{tcolorbox}
\end{frame}
