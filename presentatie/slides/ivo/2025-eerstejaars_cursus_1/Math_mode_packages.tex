\begin{frame}[fragile]{Math mode packages}
De onderstaande packages zijn belangrijk voor wiskunde en 
worden in vrijwel elk document gebruikt.
\begin{minted}[highlightlines={2,3}]{tex}
\documentclass[a4paper, 10pt]{article}
\usepackage{amsmath}
\usepackage{amssymb}
\begin{document}
Dit voegt een aantal symbolen toe
en meer opties voor de opmaak in wiskunde.
\end{document}
\end{minted}
\end{frame}
