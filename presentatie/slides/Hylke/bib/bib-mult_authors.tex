\begin{frame}[fragile]{Bibfile-meerdere auteurs}
	
	In je \mintinline{tex}{.bib}-bestand scheid je auteurs met \texttt{and}:
	\begin{minted}{bib}
		author = {A. Smith and B. Doe and E. Dropper and F. Laker}
	\end{minted}
	Zo kan \mintinline{tex}{biblatex} controleren hoeveel auteurs het toont.
	
\end{frame}
